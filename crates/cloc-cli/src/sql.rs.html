001   //! `--sql`: results as SQL statements, for loading into SQLite or Oracle.  
   002   //!  
   003   //! One row per file rather than per language, so the database can be asked  
   004   //! questions the text report cannot answer.  

   001   use crate::report::Report;  
   002   use cloc_lang::LangDb;  
   003   use std::fmt::Write as _;  
   004   use std::path::Path;  

   005   /// Which dialect to emit.  
   005   #[derive(Debug, Clone, Copy, PartialEq, Eq)]  
   006   pub enum SqlStyle {  
   006       /// SQLite and friends: `begin transaction` ... `commit`.  
   007       Default,  
   007       /// As above, but each INSERT names its columns.  
   008       NamedColumns,  
   008       /// Oracle: `TO_TIMESTAMP`, `VARCHAR2`, no transaction wrapper.  
   009       Oracle,  
   010   }  

   011   impl SqlStyle {  
   012       pub fn parse(name: &str) -> Option<Self> {  
   013           Some(match name.to_lowercase().as_str() {  
   014               "" | "default" | "sqlite" => Self::Default,  
   015               "named_columns" | "named-columns" => Self::NamedColumns,  
   016               "oracle" => Self::Oracle,  
   017               _ => return None,  
   018           })  
   019       }  
   020   }  

   009   /// Options for the SQL writer.  
   021   #[derive(Debug, Clone)]  
   022   pub struct SqlOptions {  
   023       pub style: SqlStyle,  
   010       /// `--sql-append`: add rows to an existing file, so no schema is emitted.  
   024       pub append: bool,  
   011       /// `--sql-project`: the identifier rows are tagged with.  
   025       pub project: String,  
   012       /// Seconds the run took, recorded in the metadata row.  
   026       pub elapsed_secs: f64,  
   013       /// Unix timestamp used as the run's id.  
   027       pub id: i64,  
   014       /// Human-readable timestamp for the metadata row.  
   028       pub timestamp: String,  
   029   }  

   015   /// Render a report as SQL.  
   030   pub fn render(report: &Report, db: &LangDb, opts: &SqlOptions) -> String {  
   031       let mut out = String::new();  

   016       // With --sql-append the tables already exist; emitting the schema again  
   017       // would fail the load.  
   032       if !opts.append {  
   033           out.push_str(schema(opts.style));  
   034       }  

   035       match opts.style {  
   036           SqlStyle::Oracle => {  
   037               let _ = writeln!(  
   038                   out,  
   039                   "insert into metadata values({}, TO_TIMESTAMP('{}','yyyy-mm-dd hh24:mi:ss'), '{}', {:.6});",  
   040                   opts.id,  
   041                   opts.timestamp,  
   042                   quote(&opts.project),  
   043                   opts.elapsed_secs,  
   044               );  
   045           }  
   046           SqlStyle::NamedColumns => out.push_str("begin transaction;\n"),  
   047           SqlStyle::Default => {  
   048               out.push_str("begin transaction;\n");  
   049               let _ = writeln!(  
   050                   out,  
   051                   "insert into metadata values({}, '{}', '{}', {:.6});",  
   052                   opts.id,  
   053                   opts.timestamp,  
   054                   quote(&opts.project),  
   055                   opts.elapsed_secs,  
   056               );  
   057           }  
   058       }  

   059       let insert = match opts.style {  
   060           SqlStyle::NamedColumns => {  
   061               "insert into t (id, Project, Language, File, File_dirname, \  
   062                File_basename, nBlank, nComment, nCode, nScaled ) values"  
   063           }  
   064           _ => "insert into t values",  
   065       };  

   066       for (n, entry) in report.files_by_code().iter().enumerate() {  
   067           let path = entry.path.to_string_lossy();  
   068           let dirname = entry  
   069               .path  
   070               .parent()  
   071               .map(|p| p.to_string_lossy().into_owned())  
   072               .filter(|s| !s.is_empty())  
   073               .unwrap_or_else(|| ".".to_string());  
   074           let basename = entry  
   075               .path  
   076               .file_name()  
   077               .map(|p| p.to_string_lossy().into_owned())  
   078               .unwrap_or_default();  
   018           // The scaled figure is code weighted by the language's COCOMO-style  
   019           // factor, which is what makes languages comparable.  
   079           let scaled = entry.counts.code as f64 * db.scale_factor(&entry.language).unwrap_or(1.0);  

   080           let _ = writeln!(  
   081               out,  
   082               "{insert}({}, '{}', '{}', '{}', '{}', '{}', {}, {}, {}, {:.6});",  
   083               opts.id,  
   084               quote(&opts.project),  
   085               quote(&entry.language),  
   086               quote(&path),  
   087               quote(&dirname),  
   088               quote(&basename),  
   089               entry.counts.blank,  
   090               entry.counts.comment,  
   091               entry.counts.code,  
   092               scaled,  
   093           );  

   020           // Long runs are broken into transactions so a failure does not  
   021           // discard everything.  
   094           if opts.style != SqlStyle::Oracle && (n + 1) % 10_000 == 0 {  
   095               out.push_str("commit;\nbegin transaction;\n");  
   096           }  
   097       }  

   098       if opts.style != SqlStyle::Oracle {  
   099           out.push_str("commit;\n");  
   100       }  
   101       out  
   102   }  

   022   /// Double any embedded single quote, which is how SQL escapes it.  
   103   fn quote(s: &str) -> String {  
   104       s.replace('\'', "''")  
   105   }  

   106   fn schema(style: SqlStyle) -> &'static str {  
   107       match style {  
   108           SqlStyle::Oracle => {  
   109               "  
   110   CREATE TABLE metadata  
   111   (  
   112     id          INTEGER PRIMARY KEY,  
   113     timestamp   TIMESTAMP,  
   114     project     VARCHAR2(500 CHAR),  
   115     elapsed_s   NUMBER(10, 6)  
   116   )  
   117   /  

   118   CREATE TABLE t  
   119   (  
   120     id             INTEGER           ,  
   121     project        VARCHAR2(500 CHAR),  
   122     language       VARCHAR2(500 CHAR),  
   123     file_fullname  VARCHAR2(500 CHAR),  
   124     file_dirname   VARCHAR2(500 CHAR),  
   125     file_basename  VARCHAR2(500 CHAR),  
   126     nblank         INTEGER,  
   127     ncomment       INTEGER,  
   128     ncode          INTEGER,  
   129     nscaled        NUMBER(10, 6),  
   130   FOREIGN KEY (id)  
   131       REFERENCES metadata (id)  
   132   )  
   133   /  

   134   "  
   135           }  
   136           _ => {  
   137               "  
   138   create table metadata (  
   139                   id        integer primary key,  
   140                   timestamp varchar(500),  
   141                   Project   varchar(500),  
   142                   elapsed_s real);  
   143   create table t        (  
   144                   id            integer        ,  
   145                   Project       varchar(500)   ,  
   146                   Language      varchar(500)   ,  
   147                   File          varchar(500)   ,  
   148                   File_dirname  varchar(500)   ,  
   149                   File_basename varchar(500)   ,  
   150                   nBlank        integer        ,  
   151                   nComment      integer        ,  
   152                   nCode         integer        ,  
   153                   nScaled       real           ,  
   154           foreign key (id)  
   155               references metadata (id));  
   156   "  
   157           }  
   158       }  
   159   }  

   023   /// A project identifier defaulting to the working directory, as cloc does.  
   160   pub fn default_project() -> String {  
   161       std::env::current_dir()  
   162           .map(|p| p.to_string_lossy().into_owned())  
   163           .unwrap_or_default()  
   164   }  

   024   /// `dirname` for display, matching the original's use of File::Basename.  
   165   pub fn dirname_of(path: &Path) -> String {  
   166       path.parent()  
   167           .map(|p| p.to_string_lossy().into_owned())  
   168           .filter(|s| !s.is_empty())  
   169           .unwrap_or_else(|| ".".to_string())  
   170   }  

   171   #[cfg(test)]  
   172   mod tests {  
   173       use super::*;  
   174       use crate::report::FileEntry;  
   175       use cloc_core::counter::Counts;  
   176       use std::path::PathBuf;  

   177       fn sample() -> Report {  
   178           Report {  
   179               files: vec![FileEntry {  
   180                   path: PathBuf::from("src/main.rs"),  
   181                   language: "Rust".to_string(),  
   182                   counts: Counts { blank: 2, comment: 3, code: 10 },  
   183               }],  
   184               ignored: vec![],  
   185               elapsed_secs: 0.5,  
   186           }  
   187       }  

   188       fn opts(style: SqlStyle, append: bool) -> SqlOptions {  
   189           SqlOptions {  
   190               style,  
   191               append,  
   192               project: "proj".to_string(),  
   193               elapsed_secs: 0.5,  
   194               id: 1700000000,  
   195               timestamp: "2026-07-27 12:00:00".to_string(),  
   196           }  
   197       }  

   198       #[test]  
   199       fn default_style_wraps_inserts_in_a_transaction() {  
   200           let out = render(&sample(), LangDb::default_db(), &opts(SqlStyle::Default, false));  
   201           assert!(out.contains("create table metadata"));  
   202           assert!(out.contains("begin transaction;"));  
   203           assert!(out.contains("insert into t values("));  
   204           assert!(out.trim_end().ends_with("commit;"));  
   205       }  

   025       /// --sql-append adds rows to an existing database, so re-emitting the  
   026       /// schema would break the load.  
   206       #[test]  
   207       fn append_omits_the_schema() {  
   208           let out = render(&sample(), LangDb::default_db(), &opts(SqlStyle::Default, true));  
   209           assert!(!out.contains("create table"));  
   210           assert!(out.contains("insert into t values("));  
   211       }  

   212       #[test]  
   213       fn named_columns_lists_the_columns() {  
   214           let out = render(  
   215               &sample(),  
   216               LangDb::default_db(),  
   217               &opts(SqlStyle::NamedColumns, false),  
   218           );  
   219           assert!(out.contains("insert into t (id, Project, Language, File,"));  
   220       }  

   221       #[test]  
   222       fn oracle_uses_its_own_schema_and_no_transactions() {  
   223           let out = render(&sample(), LangDb::default_db(), &opts(SqlStyle::Oracle, false));  
   224           assert!(out.contains("VARCHAR2(500 CHAR)"));  
   225           assert!(out.contains("TO_TIMESTAMP"));  
   226           assert!(!out.contains("begin transaction"));  
   227       }  

   027       /// A quote in a path would otherwise end the string literal early.  
   228       #[test]  
   229       fn single_quotes_are_doubled() {  
   230           assert_eq!(quote("it's"), "it''s");  
   231           let mut report = sample();  
   232           report.files[0].path = PathBuf::from("src/o'brien.rs");  
   233           let out = render(&report, LangDb::default_db(), &opts(SqlStyle::Default, false));  
   234           assert!(out.contains("o''brien.rs"));  
   235       }  

   028       /// The scaled column is code weighted by the language's factor.  
   236       #[test]  
   237       fn scaled_column_applies_the_language_factor() {  
   238           let db = LangDb::default_db();  
   239           let factor = db.scale_factor("Rust").unwrap();  
   240           let out = render(&sample(), db, &opts(SqlStyle::Default, false));  
   241           assert!(out.contains(&format!("{:.6}", 10.0 * factor)));  
   242       }
